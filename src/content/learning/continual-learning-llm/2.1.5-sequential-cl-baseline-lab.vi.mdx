### Task Registry & Pipeline tiền xử lý

Quản lý đặc tả 4 dataset (DBpedia, Amazon, Yahoo, AG News), hàm nạp dữ liệu tuần tự (`load_task_stream`), tokenizer và cơ chế định dạng prompt kèm mặt nạ nhãn (label loss masking):

```python
# Task Registry: Đặc tả cấu trúc cột, nhãn và cách trích xuất text của từng dataset
TASK_SPECS = {
    "dbpedia": {
        "dataset": "fancyzhx/dbpedia_14",
        "train_split": "train",
        "eval_split": "test",
        "labels": [
            "company", "school", "artist", "athlete", "politics",
            "transport", "building", "nature", "village", "animal",
            "plant", "album", "film", "book",
        ],
        "text": lambda row: f'{row["title"]}. {row["content"]}',
        "label": lambda row: row["label"],
    },
    "amazon": {
        "dataset": "fancyzhx/amazon_polarity",
        "train_split": "train",
        "eval_split": "test",
        "labels": ["negative", "positive"],
        "text": lambda row: f'{row["title"]}. {row["content"]}',
        "label": lambda row: row["label"],
    },
    "yahoo": {
        "dataset": "community-datasets/yahoo_answers_topics",
        "train_split": "train",
        "eval_split": "test",
        "labels": [
            "society", "science", "health", "education", "computer",
            "sports", "business", "entertainment", "relationships", "politics",
        ],
        "text": lambda row: " ".join(
            str(row[field] or "")
            for field in ["question_title", "question_content", "best_answer"]
        ),
        "label": lambda row: row["topic"],
    },
    "ag_news": {
        "dataset": "fancyzhx/ag_news",
        "train_split": "train",
        "eval_split": "test",
        "labels": ["world", "sports", "business", "technology"],
        "text": lambda row: row["text"],
        "label": lambda row: row["label"],
    },
}

def stable_id(task_id, split, text):
    """Tạo mã hash SHA-1 rút gọn (12 ký tự) làm khóa định danh mẫu bất biến."""
    value = f"{task_id}|{split}|{text}".encode("utf-8")
    return hashlib.sha1(value).hexdigest()[:12]

def sample_split(task_id, split_name, count, seed):
    """Trích xuất và định dạng count mẫu ngẫu nhiên từ split của dataset."""
    spec = TASK_SPECS[task_id]
    ds = load_dataset(spec["dataset"], split=spec[split_name])
    shuffled = ds.shuffle(seed=seed).select(range(count))

    rows = []
    for row in tqdm(
        shuffled,
        total=count,
        desc=f"  Tải {task_id}:{split_name}",
        leave=False,
    ):
        text = " ".join(spec["text"](row).split())[:1200]
        rows.append({
            "task_id": task_id,
            "sample_id": stable_id(task_id, split_name, text),
            "text": text,
            "label": int(spec["label"](row)),
        })
    return rows

def load_task_stream(task_order, train_per_task=2048, eval_per_task=64):
    """Nạp chuỗi dữ liệu train/eval cho toàn bộ task stream và trả về DataFrame tổng hợp."""
    data = {}
    for task_index, task_id in enumerate(task_order):
        data[task_id] = {
            "train": sample_split(
                task_id, "train_split", train_per_task, SEED + task_index
            ),
            "eval": sample_split(
                task_id, "eval_split", eval_per_task, SEED + 100 + task_index
            ),
        }

    summary_df = pd.DataFrame([
        {
            "order": f"Task {chr(65 + task_index)}",
            "task_id": task_id,
            "dataset": TASK_SPECS[task_id]["dataset"],
            "train": len(data[task_id]["train"]),
            "held_out": len(data[task_id]["eval"]),
            "labels": len(TASK_SPECS[task_id]["labels"]),
        }
        for task_index, task_id in enumerate(task_order)
    ])
    return data, summary_df

tokenizer = None

def get_tokenizer():
    """Khởi tạo tokenizer dạng lazy loading từ MODEL_NAME đã khai báo."""
    global tokenizer
    if tokenizer is None:
        tokenizer = AutoTokenizer.from_pretrained(MODEL_NAME)
        if tokenizer.pad_token_id is None:
            tokenizer.pad_token = tokenizer.eos_token
    return tokenizer

def build_prompt(task_id, text):
    """Tạo prompt dạng instruction chuẩn hóa cho bài toán phân loại."""
    options = ", ".join(TASK_SPECS[task_id]["labels"])
    target = "sentiment" if task_id == "amazon" else "topic"
    return (
        f"Instruction: Classify the {target} of the text.\n"
        f"Options: {options}. Choose exactly one option.\n"
        f"Text: {text}\n"
        "Answer:"
    )

def encode_answer(task_id, text, label_text, max_length=None):
    """Tokenize chuỗi văn bản và tạo nhãn loss masking (-100 cho toàn bộ prompt)."""
    tok = get_tokenizer()
    if max_length is None:
        max_length = globals().get("MAX_LENGTH", 192)
    target = "sentiment" if task_id == "amazon" else "topic"
    options = ", ".join(TASK_SPECS[task_id]["labels"])
    prefix = (
        f"Instruction: Classify the {target} of the text.\n"
        f"Options: {options}. Choose exactly one option.\n"
        f"Text: "
    )
    suffix = "\nAnswer:"

    prefix_ids = tok(prefix, add_special_tokens=False)["input_ids"]
    suffix_ids = tok(suffix, add_special_tokens=False)["input_ids"]
    answer_ids = tok(
        f" {label_text}{tok.eos_token}",
        add_special_tokens=False,
    )["input_ids"]

    # Cắt gọn text nếu vượt quá max_length để đảm bảo không bị tràn token
    max_text_len = max_length - len(prefix_ids) - len(suffix_ids) - len(answer_ids)
    text_ids = tok(
        text,
        add_special_tokens=False,
        truncation=True,
        max_length=max(max_text_len, 16),
    )["input_ids"]

    prompt_ids = prefix_ids + text_ids + suffix_ids
    input_ids = prompt_ids + answer_ids
    return {
        "input_ids": input_ids,
        "attention_mask": [1] * len(input_ids),
        "labels": [-100] * len(prompt_ids) + answer_ids,
    }

def encode_train_row(row):
    """Mã hóa một dòng dữ liệu huấn luyện tương ứng với nhãn đúng của nó."""
    labels = TASK_SPECS[row["task_id"]]["labels"]
    return encode_answer(row["task_id"], row["text"], labels[row["label"]])

def collate_batch(items):
    """Gom các mẫu thành batch tensor và chèn padding động theo chuỗi dài nhất."""
    tok = get_tokenizer()
    width = max(len(item["input_ids"]) for item in items)
    output = {"input_ids": [], "attention_mask": [], "labels": []}
    for item in items:
        pad = width - len(item["input_ids"])
        output["input_ids"].append(
            item["input_ids"] + [tok.pad_token_id] * pad
        )
        output["attention_mask"].append(item["attention_mask"] + [0] * pad)
        output["labels"].append(item["labels"] + [-100] * pad)
    return {key: torch.tensor(value) for key, value in output.items()}
```

### Kiến trúc mô hình & Vòng lặp huấn luyện

Khởi tạo mô hình causal language model 360M và thực hiện vòng lặp huấn luyện chuẩn PyTorch với mixed precision (FP16), gradient accumulation và gradient clipping:

```python
def load_fresh_model():
    """Khởi tạo mô hình với FP32 master weights để huấn luyện chuẩn PyTorch AMP (FP16 autocast + GradScaler)."""
    model = AutoModelForCausalLM.from_pretrained(
        MODEL_NAME,
        torch_dtype=torch.float32,
    )
    model.config.use_cache = False
    model.gradient_checkpointing_enable()
    return model.to(DEVICE)

class LiveTrainingMonitor:
    """Theo dõi và cập nhật duy nhất 1 biểu đồ loss tại chỗ (in-place) theo thời gian thực."""

    def __init__(self, method_name, task_order, total_steps_per_stage=128, update_interval=8):
        self.method_name = method_name.upper()
        self.task_order = task_order
        self.total_steps_per_stage = total_steps_per_stage
        self.update_interval = update_interval
        self.step_history = []
        self.loss_history = []
        self.stage_markers = []
        self.stage_logs = []
        self.current_step = 0
        self.display_handle = None

    def start_stage(self, stage_index, task_id):
        label = f"Stage {chr(65 + stage_index)}: {task_id}"
        self.stage_markers.append((self.current_step, label))

    def record_step(self, loss_val):
        self.current_step += 1
        self.step_history.append(self.current_step)
        self.loss_history.append(loss_val)
        if self.current_step % self.update_interval == 0:
            self.render()

    def add_stage_log(self, text):
        self.stage_logs.append(text)
        print(text)

    def render(self):
        if not self.loss_history:
            return

        fig, ax = plt.subplots(figsize=(9, 3.6))

        color_primary = "#2563EB" if self.method_name == "SEQFT" else "#0D9488"
        color_dark = "#1E40AF" if self.method_name == "SEQFT" else "#0F766E"

        ax.plot(self.step_history, self.loss_history, color=color_primary, alpha=0.3, linewidth=1.0, label="Batch Loss")
        if len(self.loss_history) > 4:
            ema = pd.Series(self.loss_history).ewm(span=8).mean().values
            ax.plot(self.step_history, ema, color=color_dark, linewidth=2.0, label="Smoothed Loss (EMA)")

        max_loss = max(self.loss_history) if self.loss_history else 4.0
        for step, label in self.stage_markers:
            if step > 0:
                ax.axvline(x=step, color="#94A3B8", linestyle="--", linewidth=1.0, alpha=0.7)
            ax.text(step + 3, max_loss * 0.9, label, fontsize=8.5, fontweight="semibold", color="#475569")

        ax.set_title(f"Live Training Loss Trajectory · [{self.method_name}]", fontsize=11, fontweight="bold", pad=10)
        ax.set_xlabel("Optimizer Updates", fontsize=9.5)
        ax.set_ylabel("Loss", fontsize=9.5)
        ax.set_xlim(0, max(len(self.task_order) * self.total_steps_per_stage, self.current_step + 10))
        ax.set_ylim(bottom=0.0)

        # Clean Style: Ẩn trục trên & trục phải
        ax.spines["top"].set_visible(False)
        ax.spines["right"].set_visible(False)
        ax.spines["left"].set_color("#94A3B8")
        ax.spines["bottom"].set_color("#94A3B8")
        ax.yaxis.grid(True, linestyle="--", alpha=0.35, color="#94A3B8")
        ax.xaxis.grid(False)
        ax.tick_params(colors="#334155")
        ax.legend(frameon=False, fontsize=8.5, loc="upper right")
        plt.tight_layout()

        # Cập nhật tại chỗ (in-place update): Đảm bảo DUY NHẤT 1 biểu đồ duy trì xuyên suốt, không tạo thêm hình mới
        if self.display_handle is None:
            try:
                self.display_handle = display(fig, display_id=True)
            except Exception:
                clear_output(wait=True)
                display(fig)
        else:
            try:
                self.display_handle.update(fig)
            except Exception:
                clear_output(wait=True)
                display(fig)

        plt.close(fig)  # Đóng figure giải phóng bộ nhớ, tránh chồng chất và cảnh báo bộ nhớ

def train_stage(
    model,
    rows,
    stage_seed,
    lr=5e-5,
    epochs=1,
    batch_size=4,
    grad_accum_steps=4,
    monitor=None,
):
    """Huấn luyện mô hình trên tập dữ liệu của một stage với FP16 AMP, Gradient Accumulation và Live Monitor."""
    seed_everything(stage_seed)
    encoded = [encode_train_row(row) for row in rows]
    loader = DataLoader(
        encoded,
        batch_size=batch_size,
        shuffle=True,
        collate_fn=collate_batch,
    )
    optimizer = torch.optim.AdamW(model.parameters(), lr=lr)
    scaler = torch.amp.GradScaler("cuda", enabled=USE_AMP)

    if torch.cuda.is_available():
        torch.cuda.reset_peak_memory_stats()
    started = time.perf_counter()
    model.train()
    optimizer.zero_grad(set_to_none=True)

    running_loss = 0.0
    update_count = 0
    progress_bar = tqdm(loader, desc="  Training", leave=False)
    for _ in range(epochs):
        for batch_index, batch in enumerate(progress_bar):
            batch = {key: value.to(DEVICE) for key, value in batch.items()}
            with torch.amp.autocast(
                "cuda", dtype=torch.float16, enabled=USE_AMP
            ):
                loss = model(**batch).loss / grad_accum_steps
            scaler.scale(loss).backward()

            loss_val = loss.item() * grad_accum_steps
            running_loss += loss_val

            is_update = (batch_index + 1) % grad_accum_steps == 0
            is_last = batch_index + 1 == len(loader)
            if is_update or is_last:
                scaler.unscale_(optimizer)
                torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
                scaler.step(optimizer)
                scaler.update()
                optimizer.zero_grad(set_to_none=True)
                update_count += 1
                if monitor is not None:
                    monitor.record_step(loss_val)
            progress_bar.set_postfix(
                step=f"{update_count}/{len(loader) // grad_accum_steps}",
                loss=f"{loss_val:.3f}",
            )

    elapsed = time.perf_counter() - started
    peak_gb = (
        torch.cuda.max_memory_allocated() / 1024**3
        if torch.cuda.is_available()
        else 0.0
    )
    avg_loss = running_loss / max(len(loader) * epochs, 1)
    return {
        "examples": len(rows),
        "non_padding_tokens": sum(
            sum(item["attention_mask"]) for item in encoded
        ) * epochs,
        "updates": update_count,
        "seconds": round(elapsed, 1),
        "peak_gpu_gb": round(peak_gb, 2),
        "avg_loss": round(avg_loss, 4),
    }
```

### Bộ đánh giá Log-probability

Đánh giá độ chính xác phân loại bằng điểm log-probability trên tập các nhãn hợp lệ, tránh sinh từ tự do (zero-generation) để đảm bảo kết quả đo lường khách quan:

```python
@torch.no_grad()
def evaluate_task(model, task_id, rows):
    """Đánh giá accuracy của mô hình trên tập mẫu rows bằng xếp hạng điểm log-probability."""
    model.eval()
    candidates = []
    gold = []

    # Tạo cặp prompt + từng nhãn ứng viên khả dĩ
    for example_index, row in enumerate(rows):
        gold.append(row["label"])
        for label_index, label_text in enumerate(TASK_SPECS[task_id]["labels"]):
            encoded = encode_answer(task_id, row["text"], label_text)
            encoded["example_index"] = example_index
            encoded["label_index"] = label_index
            candidates.append(encoded)

    def collate_candidates(items):
        batch = collate_batch(items)
        batch["example_index"] = torch.tensor(
            [item["example_index"] for item in items]
        )
        batch["label_index"] = torch.tensor(
            [item["label_index"] for item in items]
        )
        return batch

    scores = {}
    loader = DataLoader(
        candidates,
        batch_size=64,
        shuffle=False,
        collate_fn=collate_candidates,
    )
    for batch in tqdm(loader, desc=f"  Eval {task_id}", leave=False):
        example_ids = batch.pop("example_index").tolist()
        label_ids = batch.pop("label_index").tolist()
        batch = {key: value.to(DEVICE) for key, value in batch.items()}

        # Đánh giá bằng FP16 autocast để tăng tốc độ tính toán gấp đôi
        with torch.amp.autocast("cuda", dtype=torch.float16, enabled=USE_AMP):
            logits = model(
                input_ids=batch["input_ids"],
                attention_mask=batch["attention_mask"],
            ).logits[:, :-1]
        targets = batch["labels"][:, 1:]

        # Chỉ tính log-probability trên các token thuộc nhãn trả lời, bỏ qua prompt
        answer_mask = targets.ne(-100)
        safe_targets = targets.masked_fill(~answer_mask, 0)
        token_logp = logits.log_softmax(-1).gather(
            -1, safe_targets.unsqueeze(-1)
        ).squeeze(-1)
        mean_logp = (
            (token_logp * answer_mask).sum(-1)
            / answer_mask.sum(-1).clamp_min(1)
        )

        for example_id, label_id, score in zip(
            example_ids, label_ids, mean_logp.tolist()
        ):
            scores[(example_id, label_id)] = score

    # Dự đoán nhãn có điểm log-probability trung bình cao nhất
    predictions = []
    for example_index in range(len(rows)):
        label_scores = [
            scores[(example_index, label_index)]
            for label_index in range(len(TASK_SPECS[task_id]["labels"]))
        ]
        predictions.append(int(np.argmax(label_scores)))
    return float(np.mean(np.array(predictions) == np.array(gold)))

def evaluate_all(model, task_order, task_data):
    """Đánh giá mô hình trên tập held-out eval của tất cả các task trong task stream."""
    return [
        evaluate_task(model, task_id, task_data[task_id]["eval"])
        for task_id in task_order
    ]
```

### Bộ đệm Replay & Các hàm huấn luyện Baseline

Quản lý bộ đệm cân bằng cho Experience Replay và cung cấp 2 hàm huấn luyện riêng biệt: `run_seqft` cho Sequential Fine-Tuning và `run_exp_replay` cho Experience Replay:

def compose_stage_rows(current_rows, memory_rows, replay_ratio, seed):
    """Trộn mẫu của task hiện tại với mẫu cũ lấy từ bộ đệm theo tỉ lệ replay_ratio."""
    rng = random.Random(seed)
    budget = len(current_rows)
    if not memory_rows:
        return rng.sample(current_rows, k=budget)

    old_count = round(budget * replay_ratio)
    new_count = budget - old_count
    return (
        rng.sample(current_rows, k=new_count)
        + rng.choices(memory_rows, k=old_count)
    )

def update_balanced_buffer(memory_by_task, task_id, current_rows, seen_tasks, buffer_capacity, seed):
    """Cập nhật bộ đệm: phân chia đều dung lượng buffer_capacity cho tất cả các task đã học."""
    rng = random.Random(seed)
    memory_by_task[task_id] = list(current_rows)
    quota = buffer_capacity // len(seen_tasks)
    remainder = buffer_capacity % len(seen_tasks)

    balanced = {}
    for index, seen_task in enumerate(seen_tasks):
        task_quota = quota + int(index < remainder)
        source = memory_by_task[seen_task]
        balanced[seen_task] = rng.sample(
            source, k=min(task_quota, len(source))
        )
    return balanced

def flatten_buffer(memory_by_task):
    """Gom toàn bộ mẫu từ các ngăn task của bộ đệm thành danh sách phẳng để sampling."""
    return [
        row
        for task_rows in memory_by_task.values()
        for row in task_rows
    ]

Chuyển đổi mảng accuracy thành ma trận đánh giá <InlineMath formula="R" /> theo từng checkpoint và tính các chỉ số Continual Learning (Final Average Accuracy, BWT, FWT, Peak Forgetting):

def matrix_frame(results, method, task_order):
    """Chuyển đổi mảng kết quả đánh giá thành DataFrame ma trận R (hàng: checkpoint, cột: task)."""
    stage_names = ["Base"] + [
        f"After {chr(65 + index)}" for index in range(len(task_order))
    ]
    task_names = [
        f"{chr(65 + index)}: {task_id}"
        for index, task_id in enumerate(task_order)
    ]
    return pd.DataFrame(
        results[method]["matrix"],
        index=stage_names,

    )

def compute_cl_metrics(frame, task_order):
    """Tính toán 5 chỉ số Continual Learning tiêu chuẩn từ ma trận đánh giá R."""
    R = frame.to_numpy()
    task_count = len(task_order)

    # 1. Độ chính xác trung bình ở checkpoint cuối cùng
    final_average = R[-1].mean()

    # 2. Khả năng tiếp thu: điểm số ngay sau khi vừa học xong từng task (đường chéo)
    learned_once = np.array([
        R[task_index + 1, task_index]
        for task_index in range(task_count)
    ])

    # 3. Backward Transfer: mức sụt giảm trung bình so với thời điểm vừa học xong
    bwt = np.mean([
        R[-1, task_index] - R[task_index + 1, task_index]
        for task_index in range(task_count - 1)
    ])

    # 4. Forward Transfer: khả năng tổng quát hóa sang task chưa học so với Base model
    fwt = np.mean([
        R[task_index, task_index] - R[0, task_index]
        for task_index in range(1, task_count)
    ])

    # 5. Peak Forgetting: khoảng cách từ đỉnh cao nhất từng đạt được tới checkpoint cuối
    peak_forgetting = np.mean([
        R[task_index + 1:, task_index].max() - R[-1, task_index]
        for task_index in range(task_count - 1)
    ])

    return {
        "Final Average Accuracy": final_average,
        "Mean Acquisition": learned_once.mean(),
        "BWT (diagonal)": bwt,
        "FWT vs Base": fwt,
        "Peak Forgetting": peak_forgetting,
    }

    ax.set_ylim(bottom=0.0)

    # Clean style
    ax.spines["top"].set_visible(False)
    ax.spines["right"].set_visible(False)
    ax.spines["left"].set_color("#94A3B8")
    ax.spines["bottom"].set_color("#94A3B8")
    ax.yaxis.grid(True, linestyle="--", alpha=0.35, color="#94A3B8")
    ax.xaxis.grid(False)
    ax.tick_params(colors="#334155")
    ax.legend(frameon=False, fontsize=9.5, loc="upper right")

    plt.tight_layout()
    plt.show()

def report_benchmark_results(results, task_order, plot=True):
    """In đồng thời bảng chi phí huấn luyện, ma trận đánh giá R, các chỉ số Continual Learning và bộ biểu đồ trực quan hóa."""
    # 1. Bảng chi phí tính toán
    costs_df = pd.DataFrame([
        {
            "Phương pháp": method.upper(),
            "Giai đoạn": f"Stage {chr(65 + i)} ({cost['stage']})",
            "Mẫu mới": cost["current_examples"],
            "Mẫu Replay": cost["replay_examples"],
            "Loss TB": cost.get("avg_loss", "-"),
            "Updates": cost["updates"],
            "Thời gian (s)": cost["seconds"],
            "VRAM (GB)": cost["peak_gpu_gb"],
        }
        for method in results.keys()
        for i, cost in enumerate(results[method]["costs"])
    ])

    # 2. Ma trận kết quả R cho từng baseline
    matrices = {
        method: matrix_frame(results, method, task_order)
        for method in results.keys()
    }

    # 3. Tổng hợp các chỉ số Continual Learning
    metrics_df = pd.DataFrame({
        method.upper(): compute_cl_metrics(matrices[method], task_order)
        for method in results.keys()
    }).T

    print("=== 1. BẢNG CHI PHÍ HUẤN LUYỆN ===")
    display(costs_df)

    print("\n=== 2. MA TRẬN KẾT QUẢ R ===")
    for method, frame in matrices.items():
        display(frame.style.format("{:.3f}").set_caption(f"Ma trận R ({method.upper()})"))

    print("\n=== 3. CÁC CHỈ SỐ CONTINUAL LEARNING ===")
    display(metrics_df.style.format("{:.3f}"))

    # 4. Trực quan hóa Heatmap ma trận R và các biểu đồ Continual Learning
    if plot:
        print("\n=== 4. TRỰC QUAN HÓA MA TRẬN R (HEATMAP) ===")
        plot_cl_heatmaps(matrices, task_order)

        print("\n=== 5. BIỂU ĐỒ ĐỘNG THÁI LƯU GIỮ TRI THỨC (RETENTION CURVES) ===")
        plot_task_trajectories(matrices, task_order)

        print("\n=== 6. BIỂU ĐỒ ĐƯỜNG CONG HUẤN LUYỆN LOSS ===")
        plot_training_loss(results, task_order)

        print("\n=== 7. BIỂU ĐỒ CỘT SO SÁNH CÁC CHỈ SỐ CL ===")
        plot_cl_metrics_comparison(metrics_df)

    return {
        "costs": costs_df,
        "matrices": matrices,
        "metrics": metrics_df,
    }

### 2. Huấn luyện SeqFT và Experience Replay

Để thấy rõ sự khác biệt giữa tinh chỉnh tuần tự thông thường và cơ chế ôn tập tri thức, chúng ta lần lượt thực thi hai phương pháp:

#### Baseline 1 · Sequential Fine-Tuning (SeqFT)

SeqFT cập nhật mô hình trên từng task mới mà không lưu giữ hay ôn lại dữ liệu cũ. Thiết lập các siêu tham số huấn luyện và gọi hàm thực thi:

```python
# Cấu hình huấn luyện tuần tự (không dùng bộ đệm)
EPOCHS_PER_TASK = 1

SAVE_CHECKPOINTS = False

seqft_results = run_seqft(
    TASK_ORDER,
    task_data,
    epochs_per_task=EPOCHS_PER_TASK,
    learning_rate=LEARNING_RATE,
    save_checkpoints=SAVE_CHECKPOINTS,
)
```

#### Baseline 2 · Experience Replay (Replay)

Experience Replay duy trì một bộ đệm mẫu cũ và phân bổ 25% dữ liệu ôn tập trong mỗi stage. Thiết lập cấu hình bộ đệm và gọi hàm thực thi:

Baseline này đóng vai trò là cột mốc chuẩn. Khi bổ sung các phương pháp mới (EWC, LwF, O-LoRA...), bạn bắt buộc giữ cố định các thành phần nền tảng và chỉ thay đổi cơ chế thuật toán:

| Giữ cố định (Contract) | Phương pháp mới được thay đổi |
| :--- | :--- |
| `MODEL_NAME`, checkpoint ban đầu và tokenizer | Cách thức bảo vệ hoặc mở rộng mô hình |
| `TASK_ORDER` và định dạng prompt instruction | Hàm phạt điều hòa (EWC), mục tiêu chưng cất (LwF) hoặc module LoRA |
| Tập sample ID của train và held-out test | Dữ liệu phụ mà phương pháp được phép dùng (nhưng phải khai báo) |
| Ngân sách cập nhật optimizer (128 updates) và `MAX_LENGTH` | Cách tạo batch trong cùng một ngân sách |
| Quy trình đánh giá log-probability và công thức metric | Trạng thái (state) nội bộ của phương pháp |

Các bài lab EWC, LwF hay O-LoRA về sau chỉ cần triển khai một hàm strategy mới và ghi nhận kết quả vào dictionary chung:

```python
results["ewc"] = run_ewc_with_same_contract(...)
results["lwf"] = run_lwf_with_same_contract(...)
results["olora"] = run_olora_with_same_contract(...)
```

Nếu một phương pháp bắt buộc phải dùng thêm compute, checkpoint phụ hoặc dữ liệu cũ ngoài bộ đệm, chi phí phát sinh đó phải được ghi chép minh bạch vào artifact. "Cùng task order" chưa đủ để được coi là so sánh công bằng nếu ngân sách tính toán bị thay đổi ngầm.

## Thảo luận thực nghiệm và đánh đổi thực tế

Kết quả đối đầu thực nghiệm giữa Sequential Fine-Tuning (SeqFT) và Experience Replay mở ra bức tranh trực quan về sự vận hành của mô hình ngôn ngữ khi tiếp nhận các dòng tri thức liên tục. Những con số từ bảng chi phí, ma trận <InlineMath formula="R" /> và đường cong huấn luyện không chỉ phản ánh hiệu năng thuật toán, mà còn làm sáng tỏ các nguyên lý kỹ thuật cốt lõi trong Continual Learning.

### 1. Đánh đổi giữa Độ ổn định và Độ nhạy dẻo (Stability–Plasticity in Action)

Bài toán trung tâm của học liên tục luôn xoay quanh câu hỏi: làm thế nào để mô hình vừa đủ **nhạy dẻo (plasticity)** để tiếp thu task mới, vừa đủ **ổn định (stability)** để không xóa sổ tri thức cũ?

1. **Cơ chế gây quên ở SeqFT:**
   - Trong SeqFT, toàn bộ 100% dung lượng cập nhật của optimizer đều phục vụ cho task hiện tại. Khi chuyển từ Task B (Amazon - 2 nhãn) sang Task C (Yahoo - 10 nhãn), gradient của hàm loss mới kéo các ma trận trọng số dịch chuyển mạnh mẽ theo hướng tối ưu riêng của Task C.
   - Hậu quả là các rãnh biểu diễn dành cho Task A (DBpedia) và Task B bị phá vỡ cấu trúc. Sau khi học xong Task D, accuracy của Task C rơi tự do từ **67.2% xuống 54.7%** (sụt giảm 12.5%), và BWT âm sâu ở mức **-0.073**. SeqFT nghiêng hoàn toàn về phía Plasticity và đánh mất Stability.

2. **Hiện tượng Positive Backward Transfer ở Replay:**
   - Khi dành ra 25% ngân sách mỗi stage để ôn lại các mẫu cũ từ bộ đệm, điều kỳ diệu đã xảy ra: Task A (DBpedia) không những không bị quên mà độ chính xác còn tăng từ **96.9% lên 100.0%** ở checkpoint cuối cùng. Tương tự, Task C tăng từ **67.2% lên 71.9%**.
   - Chỉ số **Backward Transfer đạt giá trị dương (+0.026)**. Đây là bằng chứng thực nghiệm của hiện tượng Positive Backward Transfer: việc ôn tập dữ liệu cũ trong khi đồng thời tiếp thu dữ liệu mới giúp mô hình xây dựng không gian biểu diễn tổng quát hơn, liên kết các đặc trưng liên nhiệm vụ chặt chẽ hơn cả thời điểm nó chỉ học đơn lẻ task đó.
   - Quan trọng hơn, khả năng tiếp thu task mới (*Mean Acquisition*) của Replay vẫn đạt **87.9%** (so với 88.3% của SeqFT). Điều này chứng minh 25% replay là tỉ lệ hài hòa, bảo tồn tri thức cũ mà hầu như không làm tổn hại đến độ nhạy dẻo.

### 2. Giới hạn thực tế và bài toán mở rộng của Experience Replay

Mặc dù Experience Replay mang lại hiệu quả vượt trội trong bài lab 4 task, phương pháp này đối mặt với hai rào cản lớn khi triển khai vào các hệ thống LLM thực tế:

1. **Nút thắt dung lượng bộ đệm khi chuỗi task kéo dài (<InlineMath formula="N \to \infty" />):**
   - Với dung lượng bộ đệm cố định 256 mẫu, khi học 4 task, mỗi task được giữ trung bình ~64 đến 85 mẫu. Số lượng này đủ để đại diện cho phân phối dữ liệu gốc.
   - Tuy nhiên, nếu hệ thống phải học liên tục qua 20 hay 100 domain khác nhau, hạn ngạch cho mỗi task cũ sẽ bị co hẹp chỉ còn **2–5 mẫu**. Với không gian ngôn ngữ rộng lớn, vài mẫu văn bản ngắn ngủi không thể đại diện đầy đủ cho phân phối của toàn bộ một lĩnh vực, khiến hiệu quả chống quên lãng của Replay suy giảm nghiêm trọng.
   - Nếu giải quyết bằng cách tăng dung lượng bộ đệm tuyến tính theo số task, chi phí lưu trữ và thời gian truy xuất sẽ bùng nổ, biến việc học liên tục trở lại bài toán huấn luyện đa nhiệm (multi-task training) tốn kém.

2. **Rào cản pháp lý và quyền riêng tư dữ liệu (Privacy & GDPR):**
   - Trong các hệ thống sản xuất thực tế (ngân hàng, chăm sóc sức khỏe, trợ lý doanh nghiệp), dữ liệu thường gắn liền với thông tin định danh người dùng (PII).
   - Theo các quy định bảo mật như GDPR ("quyền được lãng quên"), doanh nghiệp không được phép lưu trữ vĩnh viễn các đoạn chat hay văn bản cũ của khách hàng trong bộ đệm để replay định kỳ. Khi dữ liệu quá khứ bắt buộc phải bị xóa sạch khỏi hệ thống, Replay thuần túy hoàn toàn bất khả thi.

### 3. Cầu nối sang các họ phương pháp tiếp theo trong Chương 2

Chính những giới hạn vật lý và pháp lý của Experience Replay là động lực thúc đẩy sự ra đời của các họ giải pháp Continual Learning tiên tiến hơn mà bạn sẽ khám phá ở các bài tiếp theo:

| Họ giải pháp | Cơ chế bảo vệ tri thức | Khác biệt so với Replay |
| :--- | :--- | :--- |
| **Weight Regularization (EWC)** | Tính toán ma trận thông tin Fisher để xác định trọng số nào quan trọng với task cũ, sau đó phạt nặng nếu optimizer cố tình thay đổi các trọng số này | **Không cần lưu trữ bất kỳ mẫu dữ liệu cũ nào**, bảo mật tuyệt đối dữ liệu quá khứ. |
| **Function Regularization (Distillation / LwF)** | Sử dụng checkpoint mô hình ở stage trước làm "giáo viên" (teacher), phạt sự phân kỳ trong phân phối xác suất đầu ra (logits) của mô hình hiện tại | Bảo tồn hành vi hàm ánh xạ của mô hình thay vì cố định từng trọng số cục bộ. |
| **Architecture Expansion & Isolation (O-LoRA)** | Đóng băng hoàn toàn mô hình gốc, cấp phát các module LoRA riêng biệt cho từng task và ép các cập nhật nằm trên các không gian con trực giao | Triệt tiêu hoàn toàn xung đột gradient giữa các task, loại bỏ hiện tượng can thiệp tiêu cực. |

### 4. Câu hỏi phản biện và suy ngẫm

Để khắc sâu các nguyên lý đã kiểm chứng, bạn hãy tự trả lời các câu hỏi sau trước khi bước sang bài lab tiếp theo:

1. **Đánh đổi về tỉ lệ Replay:** Nếu chúng ta tăng tỉ lệ replay từ 25% lên 50% hoặc 75%, điều gì sẽ xảy ra với chỉ số Mean Acquisition và thời gian hội tụ của task mới? Tại sao trong Continual Learning, việc tăng replay quá mức có thể dẫn đến hiện tượng "trì trệ tiếp thu" (*plasticity bottleneck*)?
2. **Ảnh hưởng của Pretraining Prior:** Tại sao trong thực nghiệm SeqFT, Task B (Amazon Polarity) hầu như không bị quên lãng (vẫn giữ vững accuracy 90.6%), trong khi Task C (Yahoo Answers) lại rớt thảm hại về 54.7%? Năng lực hiểu ngôn ngữ sẵn có từ giai đoạn pretrain của LLM đóng vai trò gì trong sự phân hóa này?
