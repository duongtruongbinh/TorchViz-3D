```python
def encode_answer(task_id, text, label_text, max_length=MAX_LENGTH):
    labels = TASK_SPECS[task_id]["labels"]
    target = "sentiment" if task_id == "amazon" else "topic"
    prefix = f"Instruction: Classify the {target} of the text.\nOptions: {', '.join(labels)}. Choose exactly one option.\nText: "
    suffix = "\nAnswer:"
    prefix_ids = tokenizer(prefix, add_special_tokens=False)["input_ids"]
    suffix_ids = tokenizer(suffix, add_special_tokens=False)["input_ids"]
    answer_ids = tokenizer(f" {label_text}{tokenizer.eos_token}", add_special_tokens=False)["input_ids"]
    max_text = max_length - len(prefix_ids) - len(suffix_ids) - len(answer_ids)
    text_ids = tokenizer(text, add_special_tokens=False, truncation=True, max_length=max(max_text, 16))["input_ids"]
    prompt_ids = prefix_ids + text_ids + suffix_ids
    input_ids = prompt_ids + answer_ids
    return {
        "input_ids": input_ids,
        "attention_mask": [1] * len(input_ids),
        "labels": [-100] * len(prompt_ids) + answer_ids,
```

## Thuật toán SeqFT và Experience Replay

Cell 3 định nghĩa các hàm tính điểm log-probability cho nhãn ứng viên, tính toán các chỉ số Continual Learning, cơ chế bộ đệm cân bằng cho Experience Replay, và vòng lặp huấn luyện chuẩn xác.

### Cell 3 · Bộ đánh giá, Replay buffer và hàm huấn luyện

```python
def load_fresh_model():
    model = AutoModelForCausalLM.from_pretrained(
        MODEL_NAME,
        revision=MODEL_REVISION,
        torch_dtype=torch.float32,
    )
    model.config.use_cache = False
    model.gradient_checkpointing_enable()
    return model.to(DEVICE)

@torch.no_grad()
def evaluate_task(model, task_id, rows):
    model.eval()
    candidates = []
    gold = []
    for example_index, row in enumerate(rows):
        gold.append(row["label"])
        for label_index, label_text in enumerate(TASK_SPECS[task_id]["labels"]):
            encoded = encode_answer(task_id, row["text"], label_text)
            encoded["example_index"] = example_index
            encoded["label_index"] = label_index
            candidates.append(encoded)

    def collate_candidates(items):
        batch = collate_batch(items)
        batch["example_index"] = torch.tensor([item["example_index"] for item in items])
        batch["label_index"] = torch.tensor([item["label_index"] for item in items])
        return batch

    scores = {}
    loader = DataLoader(candidates, batch_size=64, shuffle=False, collate_fn=collate_candidates)
    for batch in tqdm(loader, desc=f"  Eval {task_id}", leave=False):
        example_ids = batch.pop("example_index").tolist()
        label_ids = batch.pop("label_index").tolist()
        batch = {key: value.to(DEVICE) for key, value in batch.items()}
        with torch.amp.autocast("cuda", dtype=torch.float16, enabled=USE_AMP):
            logits = model(input_ids=batch["input_ids"], attention_mask=batch["attention_mask"]).logits[:, :-1]
        targets = batch["labels"][:, 1:]
        answer_mask = targets.ne(-100)
        safe_targets = targets.masked_fill(~answer_mask, 0)
        token_logp = logits.log_softmax(-1).gather(-1, safe_targets.unsqueeze(-1)).squeeze(-1)
        mean_logp = (token_logp * answer_mask).sum(-1) / answer_mask.sum(-1).clamp_min(1)
        for example_id, label_id, score in zip(example_ids, label_ids, mean_logp.tolist()):
            scores[(example_id, label_id)] = score

    predictions = [
        int(np.argmax([scores[(i, li)] for li in range(len(TASK_SPECS[task_id]["labels"]))]))
        for i in range(len(rows))
    ]
    return float(np.mean(np.array(predictions) == np.array(gold)))

def evaluate_all(model, task_order, task_data):
    return [evaluate_task(model, task_id, task_data[task_id]["eval"]) for task_id in task_order]

def compute_cl_metrics(matrix):
    R = np.asarray(matrix, dtype=float)
    task_count = R.shape[1]
    final_avg = float(R[-1].mean())
    learned_once = np.array([R[i + 1, i] for i in range(task_count)])
    learning_gains = learned_once - R[0]
    bwt = float(np.mean([R[-1, i] - R[i + 1, i] for i in range(task_count - 1)]))
    fwt = float(np.mean([R[i, i] - R[0, i] for i in range(1, task_count)]))
    peak_forgetting = float(np.mean([R[i + 1:, i].max() - R[-1, i] for i in range(task_count - 1)]))
    return {
        "final_avg": round(final_avg, 4),
        "mean_acq": round(float(learned_once.mean()), 4),
        "mean_learning_gain": round(float(learning_gains.mean()), 4),
        "min_learning_gain": round(float(learning_gains.min()), 4),
        "bwt": round(bwt, 4),
        "fwt": round(fwt, 4),
        "peak_forgetting": round(peak_forgetting, 4),
    }

def compose_stage_rows(current_rows, memory_rows, replay_ratio, seed):
    rng = random.Random(seed)
    budget = len(current_rows)
    if not memory_rows:
        return list(current_rows)
    old_count = round(budget * replay_ratio)
    new_count = budget - old_count
    return rng.sample(current_rows, k=new_count) + rng.choices(memory_rows, k=old_count)

def update_balanced_buffer(memory_by_task, task_id, current_rows, seen_tasks, buffer_capacity, seed):
    rng = random.Random(seed)
    memory_by_task[task_id] = list(current_rows)
    quota = buffer_capacity // len(seen_tasks)
    remainder = buffer_capacity % len(seen_tasks)
    balanced = {}
    for index, seen_task in enumerate(seen_tasks):
        task_quota = quota + int(index < remainder)
        source = memory_by_task[seen_task]
        balanced[seen_task] = rng.sample(source, k=min(task_quota, len(source)))
    return balanced

def flatten_buffer(memory_by_task):
    return [row for rows in memory_by_task.values() for row in rows]

def train_stage(model, rows, seed, lr=5e-5, epochs=1, batch_size=4, grad_accum_steps=4):
    rng = random.Random(seed)
    shuffled = list(rows)
    rng.shuffle(shuffled)
    encoded = [encode_train_row(row) for row in shuffled]
    loader = DataLoader(encoded, batch_size=batch_size, shuffle=False, collate_fn=collate_batch)
    optimizer = torch.optim.AdamW(model.parameters(), lr=lr, betas=(0.9, 0.999), eps=1e-8, weight_decay=0.01)
    scaler = torch.amp.GradScaler("cuda", init_scale=AMP_INITIAL_SCALE, enabled=USE_AMP)

    started = time.perf_counter()
    running_loss = 0.0
    loss_history = []
    update_loss_sum = 0.0
    update_microbatches = 0
    update_count = 0
    model.train()
    optimizer.zero_grad(set_to_none=True)
    if torch.cuda.is_available():
        torch.cuda.reset_peak_memory_stats()

    for epoch in range(epochs):
        for step, batch in enumerate(loader):
            batch = {key: value.to(DEVICE) for key, value in batch.items()}
            with torch.amp.autocast("cuda", dtype=torch.float16, enabled=USE_AMP):
                outputs = model(**batch)
                loss = outputs.loss / grad_accum_steps
            scaler.scale(loss).backward()
            raw_loss_value = float(outputs.loss.item())
            running_loss += raw_loss_value
            update_loss_sum += raw_loss_value
            update_microbatches += 1

            if (step + 1) % grad_accum_steps == 0 or (step + 1) == len(loader):
                scaler.unscale_(optimizer)
                torch.nn.utils.clip_grad_norm_(model.parameters(), 1.0)
                scale_before = scaler.get_scale()
                scaler.step(optimizer)
                scaler.update()
                if USE_AMP and scaler.get_scale() < scale_before:
                    raise RuntimeError(
                        "AMP đã bỏ qua một optimizer update. Hãy giảm AMP_INITIAL_SCALE "
                        "để giữ cùng update budget giữa các phương pháp."
                    )
                optimizer.zero_grad(set_to_none=True)
                update_count += 1
                loss_history.append(update_loss_sum / update_microbatches)
                update_loss_sum = 0.0
                update_microbatches = 0

    expected_updates = epochs * ((len(loader) + grad_accum_steps - 1) // grad_accum_steps)
    if update_count != expected_updates:
        raise RuntimeError(f"Update budget sai: expected={expected_updates}, actual={update_count}")
    elapsed = time.perf_counter() - started
    peak_gb = torch.cuda.max_memory_allocated() / 1024**3 if torch.cuda.is_available() else 0.0
    avg_loss = running_loss / max(len(loader) * epochs, 1)
    return {
        "updates": update_count,
        "seconds": round(elapsed, 1),
        "peak_gpu_gb": round(peak_gb, 2),
        "avg_loss": round(avg_loss, 4),
        "loss_history": loss_history,
    }

def run_seqft(task_order, task_data, batch_size=4, grad_accum_steps=4, epochs_per_task=1, learning_rate=5e-5):
    seed_everything(SEED)
    model = load_fresh_model()
    base_eval = evaluate_all(model, task_order, task_data)
    matrix = [base_eval]
    costs = []
    all_loss_history = []

    for stage_index, task_id in enumerate(task_order):
        stage_rows = task_data[task_id]["train"]
        cost = train_stage(
            model, stage_rows, SEED + 2000 + stage_index,
            lr=learning_rate, epochs=epochs_per_task,
            batch_size=batch_size, grad_accum_steps=grad_accum_steps,
        )
        cost_info = {
            "stage": task_id,
            "current_examples": len(stage_rows),
            "replay_examples": 0,
            "updates": cost["updates"],
            "seconds": cost["seconds"],
            "peak_gpu_gb": cost["peak_gpu_gb"],
            "avg_loss": cost["avg_loss"],
        }
        costs.append(cost_info)
        all_loss_history.extend(cost["loss_history"])
        matrix.append(evaluate_all(model, task_order, task_data))

    del model
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
    return {
        "matrix": matrix,
        "metrics": compute_cl_metrics(matrix),
        "costs": costs,
        "loss_history": all_loss_history,
    }

def run_exp_replay(task_order, task_data, buffer_capacity=256, replay_ratio=0.25, batch_size=4, grad_accum_steps=4, epochs_per_task=1, learning_rate=5e-5):
    seed_everything(SEED)
    model = load_fresh_model()
    base_eval = evaluate_all(model, task_order, task_data)
    matrix = [base_eval]
    costs = []
    all_loss_history = []
    memory_by_task = {}

    for stage_index, task_id in enumerate(task_order):
        memory_rows = flatten_buffer(memory_by_task)
        stage_rows = compose_stage_rows(task_data[task_id]["train"], memory_rows, replay_ratio, SEED + 1000 + stage_index)
        cur_cnt = sum(row["task_id"] == task_id for row in stage_rows)
        rep_cnt = sum(row["task_id"] != task_id for row in stage_rows)
        cost = train_stage(
            model, stage_rows, SEED + 2000 + stage_index,
            lr=learning_rate, epochs=epochs_per_task,
            batch_size=batch_size, grad_accum_steps=grad_accum_steps,
        )
        cost_info = {
            "stage": task_id,
            "current_examples": cur_cnt,
            "replay_examples": rep_cnt,
            "updates": cost["updates"],
            "seconds": cost["seconds"],
            "peak_gpu_gb": cost["peak_gpu_gb"],
            "avg_loss": cost["avg_loss"],
        }
        costs.append(cost_info)
        all_loss_history.extend(cost["loss_history"])
        matrix.append(evaluate_all(model, task_order, task_data))
        memory_by_task = update_balanced_buffer(
            memory_by_task, task_id, task_data[task_id]["train"],
            task_order[:stage_index + 1], buffer_capacity, SEED + 3000 + stage_index,
        )

    del model
    gc.collect()
    if torch.cuda.is_available():
        torch.cuda.empty_cache()
    return {
        "matrix": matrix,
        "metrics": compute_cl_metrics(matrix),
        "costs": costs,
        "loss_history": all_loss_history,
    }

def save_benchmark_artifact(output_dir, results, task_data, task_order):
    os.makedirs(output_dir, exist_ok=True)
    artifact = {
        "config": {
            "seed": SEED,
            "model_name": MODEL_NAME,
            "model_revision": MODEL_REVISION,
            "dataset_revisions": {
                task_id: TASK_SPECS[task_id]["revision"] for task_id in task_order
            },
            "task_order": task_order,
            "train_per_task": TRAIN_PER_TASK,
            "eval_per_task": EVAL_PER_TASK,
            "max_length": MAX_LENGTH,
            "batch_size": BATCH_SIZE,
            "grad_accum_steps": GRAD_ACCUM_STEPS,
            "learning_rate": LEARNING_RATE,
            "epochs_per_task": EPOCHS_PER_TASK,
            "buffer_capacity": BUFFER_CAPACITY,
            "replay_ratio": REPLAY_RATIO,
        },
        "sample_ids": {
            task_id: {
                split: [row["sample_id"] for row in rows]
                for split, rows in task_data[task_id].items()
            }
            for task_id in task_order
        },
        "sampled_rows": task_data,
        "results": results,
    }
    artifact_path = os.path.join(output_dir, "seqft_replay_seed42.json")
    temp_path = artifact_path + ".tmp"
    with open(temp_path, "w", encoding="utf-8") as file:
        json.dump(artifact, file, ensure_ascii=False, indent=2)
        file.flush()
        os.fsync(file.fileno())
    os.replace(temp_path, artifact_path)
    print(f"Đã lưu thành công artifact benchmark vào: {artifact_path}")
    return artifact_path
```

## Kiểm tra output và diễn giải kết quả

Code in trực tiếp cost records, hai ma trận <InlineMath formula="R" /> và bảng metric từ run hiện tại. Không dùng lại bảng số của một run cũ sau khi contract, seed handling hoặc data ordering đã thay đổi.

### 1. Kiểm tra contract trước khi đọc điểm

- Base row của SeqFT và Replay phải giống nhau vì hai run dùng cùng checkpoint và evaluator.
- Stage A chưa có memory, nên hai run phải dùng cùng dữ liệu, thứ tự và seed; hàng `After A` phải giống nhau. Nếu khác, comparison đã có nhiễu trước khi intervention bắt đầu.
- Mỗi stage phải có đúng 128 optimizer updates. Từ Stage B, Replay dùng 1.536 vị trí cho task mới và 512 vị trí cho buffer, tổng vẫn là 2.048.
- Nếu AMP bỏ qua một update, cell dừng thay vì lưu artifact có compute budget lệch.

### 2. Đọc retention cùng acquisition

Final Average Accuracy cho biết chất lượng checkpoint cuối trên bốn task. BWT và Peak Forgetting đo mức suy giảm của task cũ, còn Mean Acquisition, Mean Learning Gain và Min Learning Gain cho biết model có thực sự học task mới hay không.

Replay chỉ hỗ trợ giả thuyết retention khi BWT tiến gần 0 hơn hoặc Peak Forgetting giảm mà learning gain vẫn dương. Forgetting thấp do model không học được task mới không phải là một kết quả tốt.

### 3. Đọc ma trận theo từng task

Một metric trung bình có thể che giấu task bị quên mạnh. Hãy đọc từng cột từ diagonal checkpoint tới hàng cuối, rồi đối chiếu với số mẫu của task đó trong balanced buffer. Buffer cố định 256 mẫu sẽ chia quota nhỏ dần khi số task tăng, nên retention có thể không đồng đều.

### 4. Không dùng training loss thay cho retention

Từ Stage B, SeqFT và Replay tối ưu các batch có thành phần khác nhau. Loss thấp hơn chỉ mô tả objective nội bộ của method; bằng chứng retention phải đến từ held-out matrix và CL metrics.

### 5. Ranh giới kết luận

Run seed 42 là một fixture có kiểm soát, không ước lượng biến thiên theo seed. Replay cũng giữ raw example, nên yêu cầu privacy, consent và data-retention của hệ thống thực tế có thể giới hạn cách dùng buffer. Lab tiếp theo kiểm tra EWC và SI, hai phương pháp tổng hợp importance ở task boundary thay vì trộn raw example cũ vào từng batch mới.
