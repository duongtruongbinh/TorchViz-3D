Ảnh hóa đơn được validate dưới tenant, encode/reference theo SDK, rồi request extraction schema. Output amount phải cross-check currency/total trước automation.

~~~ts
import assert from "node:assert/strict";
type Upload = {
  owner: string; mime: "image/jpeg" | "image/png"; bytes: Uint8Array;
};
type ImageBlock = {
  type: "image";
  source: { type: "base64"; media_type: Upload["mime"]; data: string };
};

function toImageBlock(upload: Upload, subject: string): ImageBlock {
  if (upload.owner !== subject) throw new Error("forbidden_media");
  if (upload.bytes.byteLength === 0 || upload.bytes.byteLength > 5_000_000) {
    throw new Error("invalid_media_size");
  }
  return {
    type: "image",
    source: {
      type: "base64",
      media_type: upload.mime,
      data: Buffer.from(upload.bytes).toString("base64"),
    },
  };
}

const block = toImageBlock(
  { owner: "tenant-a", mime: "image/png", bytes: new Uint8Array([137, 80, 78, 71]) },
  "tenant-a",
);
assert.equal(block.source.media_type, "image/png");
assert.equal(block.source.data, "iVBORw==");
assert.throws(
  () => toImageBlock(
    { owner: "tenant-a", mime: "image/png", bytes: new Uint8Array([1]) },
    "tenant-b",
  ),
  /forbidden_media/,
);
~~~
<LessonNote>Fixture minh họa ownership/byte boundary. Production còn phải kiểm tra magic bytes, dimensions, malware policy và xóa media theo retention.</LessonNote>
