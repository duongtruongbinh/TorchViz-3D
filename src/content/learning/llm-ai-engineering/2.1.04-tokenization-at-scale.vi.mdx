  contract="manifest cố định + segment statistics theo source/language"
  output="version/hash/boundary ID và tokens-per-KiB"
  pitfall="Không đổi normalization/revision, mất boundary/EOS, hoặc chỉ đo documents/second."
  takeaway="Handoff cần manifest tái lập được và thống kê tổng hợp, không cần log raw document."

~~~ts
import assert from "node:assert/strict";

type TokenizerManifest = {
  version: string;
  normalizationHash: string;
  boundaryTokenId: number;
};
type SegmentStat = { language: string; bytes: number; tokens: number };

function assertManifest(value: TokenizerManifest): void {
  if (!value.version || !/^[a-f0-9]{8,64}$/.test(value.normalizationHash)) {
    throw new Error("tokenizer manifest is not reproducible");
  }
  if (!Number.isInteger(value.boundaryTokenId) || value.boundaryTokenId < 0) {
    throw new Error("boundaryTokenId must be a non-negative integer");
  }
}

function tokensPerKib(stat: SegmentStat): number {
  if (!stat.language || stat.bytes <= 0 || stat.tokens < 0) {
    throw new Error("invalid segment statistic");
  }
  return stat.tokens * 1024 / stat.bytes;
}

assertManifest({
  version: "support-tokenizer-v3",
  normalizationHash: "a1b2c3d4",
  boundaryTokenId: 2,
});
const compact = tokensPerKib({ language: "en", bytes: 1_000, tokens: 250 });
const expanded = tokensPerKib({ language: "vi", bytes: 1_000, tokens: 600 });
assert.equal(expanded / compact, 2.4);
assert.throws(
  () => assertManifest({
    version: "v3",
    normalizationHash: "",
    boundaryTokenId: 2,
  }),
  /not reproducible/,
);
~~~
