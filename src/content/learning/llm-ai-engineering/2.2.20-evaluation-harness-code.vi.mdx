```python
from dataclasses import dataclass
from typing import Callable

@dataclass(frozen=True)
class EvalCase:
    id: str
    prompt: str
    required_terms: tuple[str, ...]
    critical: bool = False

@dataclass(frozen=True)
class EvalResult:
    case_id: str
    passed: bool
    reason: str

def run_eval(cases: list[EvalCase], system: Callable[[str], str]) -> list[EvalResult]:
    if not cases:
        raise ValueError('cases must be non-empty')
    ids = [case.id for case in cases]
    if len(ids) != len(set(ids)):
        raise ValueError('case ids must be unique')
    results: list[EvalResult] = []
    for case in cases:
        try:
            response = system(case.prompt)
            if not isinstance(response, str):
                raise TypeError('system response must be str')
            missing = [term for term in case.required_terms if term not in response]
            results.append(EvalResult(case.id, not missing, 'ok' if not missing else f'missing:{missing}'))
        except Exception as error:
            results.append(EvalResult(case.id, False, f'error:{type(error).__name__}'))
    return results

def release_allowed(cases: list[EvalCase], results: list[EvalResult], minimum_pass_rate: float) -> bool:
    if not cases or not 0 <= minimum_pass_rate <= 1 or len(cases) != len(results):
        raise ValueError('invalid gate inputs')
    by_id = {result.case_id: result for result in results}
    if len(by_id) != len(results) or set(by_id) != {case.id for case in cases}:
        raise ValueError('result ids must uniquely match cases')
    if any(case.critical and not by_id[case.id].passed for case in cases):
        return False
    return sum(result.passed for result in results) / len(results) >= minimum_pass_rate

cases = [
    EvalCase('normal', 'status?', ('delivered',)),
    EvalCase('auth', 'refund?', ('denied',), critical=True),
]
fixture = {'status?': 'delivered', 'refund?': 'denied: confirmation required'}
results = run_eval(cases, fixture.__getitem__)
assert [result.passed for result in results] == [True, True]
assert release_allowed(cases, results, 1.0)

broken = run_eval(cases, lambda prompt: 'delivered' if prompt == 'status?' else 'approved')
assert not release_allowed(cases, broken, 0.5)
try:
    run_eval([cases[0], cases[0]], fixture.__getitem__)
except ValueError as error:
    assert 'unique' in str(error)
else:
    raise AssertionError('duplicate ids must fail')
for invalid_cases, invalid_results in [
    ([], []),
    (cases, [results[0], results[0]]),
]:
    try:
        release_allowed(invalid_cases, invalid_results, 1.0)
    except ValueError:
        pass
    else:
        raise AssertionError('empty or duplicate result sets must fail')
```
