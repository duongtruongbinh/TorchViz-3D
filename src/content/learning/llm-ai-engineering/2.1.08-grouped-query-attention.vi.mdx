Nếu `H_q=32`, `H_kv=8`, mỗi KV head phục vụ 4 query heads. So với MHA 32 KV heads, phần KV cache giảm `32/8 = 4×`; so với MQA một head, nó vẫn lớn hơn 8×.

~~~ts
import assert from "node:assert/strict";

function gqaLayout(queryHeads: number, kvHeads: number) {
  if (
    !Number.isInteger(queryHeads)
    || !Number.isInteger(kvHeads)
    || queryHeads <= 0
    || kvHeads <= 0
    || queryHeads % kvHeads !== 0
  ) {
    throw new Error("queryHeads must be divisible by positive kvHeads");
  }
  return {
    queriesPerKvHead: queryHeads / kvHeads,
    cacheReductionVsMha: queryHeads / kvHeads,
  };
}

assert.deepEqual(gqaLayout(32, 8), {
  queriesPerKvHead: 4,
  cacheReductionVsMha: 4,
});
assert.throws(() => gqaLayout(30, 8), /divisible/);
~~~
