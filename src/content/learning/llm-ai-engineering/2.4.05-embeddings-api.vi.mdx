Hai documents tạo `(2,3)` trong fixture: `[1,0,0]` và `[0.8,0.2,0]`. Query `[1,0,0]` có cosine cao hơn với document đầu. Đổi dimension cần reindex, không append vào index cũ.

~~~ts
import assert from "node:assert/strict";
type Vector = number[];
function cosine(a: Vector, b: Vector): number {
  if (a.length === 0 || a.length !== b.length) throw new Error("dimension_mismatch");
  const dot = a.reduce((sum, value, i) => sum + value * b[i], 0);
  const normA = Math.sqrt(a.reduce((sum, value) => sum + value * value, 0));
  const normB = Math.sqrt(b.reduce((sum, value) => sum + value * value, 0));
  if (normA === 0 || normB === 0) throw new Error("zero_vector");
  return dot / (normA * normB);
}

const documents: Vector[] = [[1, 0, 0], [0.8, 0.2, 0]];
const query: Vector = [1, 0, 0];
assert.equal(documents.length, 2);        // N = 2
assert.ok(documents.every(v => v.length === 3)); // D = 3
const scores = documents.map(vector => cosine(query, vector));
assert.equal(scores[0], 1);
assert.ok(scores[0] > scores[1]);
assert.throws(() => cosine(query, [1, 0]), /dimension_mismatch/);
~~~
