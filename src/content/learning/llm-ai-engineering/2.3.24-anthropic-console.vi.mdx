~~~ts
import assert from "node:assert/strict";

type ExperimentSpec = {
  id: string;
  promptVersion: string;
  modelId: string;
  schemaVersion: string;
  evalSuiteVersion: string;
  checkedOn: string;
  fixtureIds: readonly string[];
  maxOutputTokens: number;
};

function toAnthropicExperiment(spec: ExperimentSpec) {
  if (!/^\d{4}-\d{2}-\d{2}$/.test(spec.checkedOn)) throw new Error("invalid_date");
  if (spec.fixtureIds.length === 0) throw new Error("empty_fixture_set");
  if (!Number.isInteger(spec.maxOutputTokens) || spec.maxOutputTokens <= 0) {
    throw new Error("invalid_output_budget");
  }
  return {
    provider: "anthropic" as const,
    apiSurface: "Messages API",
    experimentId: spec.id,
    promptVersion: spec.promptVersion,
    modelId: spec.modelId,
    schemaVersion: spec.schemaVersion,
    evalSuiteVersion: spec.evalSuiteVersion,
    checkedOn: spec.checkedOn,
    requestShape: {
      systemFrom: spec.promptVersion,
      userMessageFixtureIds: [...spec.fixtureIds],
      expectedContent: "ordered content blocks",
      maxTokens: spec.maxOutputTokens,
    },
  };
}

const spec: ExperimentSpec = {
  id: "support-exp-17",
  promptVersion: "support-v12",
  modelId: "configured-model-id",
  schemaVersion: "support-v4",
  evalSuiteVersion: "policy-set-7",
  checkedOn: "2026-07-23",
  fixtureIds: ["ticket-01", "ticket-02", "ticket-03"],
  maxOutputTokens: 300,
};
const result = toAnthropicExperiment(spec);
assert.equal(result.apiSurface, "Messages API");
assert.equal(result.requestShape.systemFrom, "support-v12");
assert.equal(result.requestShape.expectedContent, "ordered content blocks");
assert.equal("apiKey" in result, false);
assert.throws(
  () => toAnthropicExperiment({ ...spec, fixtureIds: [] }),
  /empty_fixture_set/,
);
~~~
