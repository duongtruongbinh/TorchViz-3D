Claude trả tool_use `lookup_order`; app validate schema/ownership, chạy read-only function, gửi tool_result. Fine-grained streamed args phải buffer và parse guard.

~~~ts
import assert from "node:assert/strict";
type ToolUse = { id: string; name: string; inputFragments: string[] };
type ToolResultBlock = {
  type: "tool_result"; tool_use_id: string; content: string; is_error: boolean;
};

function runClientTool(call: ToolUse, ownedOrders: Set<string>): ToolResultBlock {
  const error = (code: string): ToolResultBlock => ({
    type: "tool_result", tool_use_id: call.id,
    content: JSON.stringify({ error: code }), is_error: true,
  });
  if (call.name !== "lookup_order") return error("unknown_tool");
  let input: unknown;
  try { input = JSON.parse(call.inputFragments.join("")); }
  catch { return error("invalid_json"); }
  const orderId = (input as { orderId?: unknown })?.orderId;
  if (typeof orderId !== "string" || !ownedOrders.has(orderId)) return error("forbidden");
  return {
    type: "tool_result", tool_use_id: call.id,
    content: '{"status":"in_transit"}', is_error: false,
  };
}

const result = runClientTool({
  id: "toolu_17", name: "lookup_order",
  inputFragments: ['{"order', 'Id":"A17"}'],
}, new Set(["A17"]));
assert.deepEqual(result, {
  type: "tool_result", tool_use_id: "toolu_17",
  content: '{"status":"in_transit"}', is_error: false,
});
assert.equal(
  runClientTool(
    { id: "toolu_18", name: "lookup_order", inputFragments: ['{"orderId":'] },
    new Set(["A17"]),
  ).is_error,
  true,
);
~~~
<LessonNote>Chỉ parse sau khi tool-use block hoàn tất. Application phải cap vòng/calls, authorize từng action và dùng idempotency cho side effect.</LessonNote>
