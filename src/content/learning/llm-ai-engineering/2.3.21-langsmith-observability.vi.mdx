<CodeLessonFrame title="Safe trace contract" contract="redacted attributes → typed SafeTrace" output="req-017 · support-v12 · 420 tokens · validation_error" pitfall="Không auto-capture prompt/response, dùng email làm trace ID hoặc bỏ version redaction policy." takeaway="Redaction và sampling xảy ra trước exporter; retention/access/region phải được kiểm tra theo deployment.">

~~~ts
import assert from "node:assert/strict";

type SafeTrace = {
  correlationId: string;
  promptVersion: string;
  modelClass: string;
  tokensIn: number;
  status: "ok" | "validation_error" | "provider_error";
  tenantPseudonym: string;
};

function buildSafeTrace(input: Record<string, unknown>): SafeTrace {
  for (const forbidden of ["prompt", "response", "email", "apiKey", "secret"]) {
    if (forbidden in input) throw new Error(`sensitive_attribute:${forbidden}`);
  }
  const required = [
    "correlationId", "promptVersion", "modelClass", "status", "tenantPseudonym",
  ] as const;
  for (const key of required) {
    if (typeof input[key] !== "string" || !(input[key] as string).trim()) {
      throw new Error(`invalid_attribute:${key}`);
    }
  }
  if (!Number.isInteger(input.tokensIn) || (input.tokensIn as number) < 0) {
    throw new Error("invalid_attribute:tokensIn");
  }
  if (!["ok", "validation_error", "provider_error"].includes(input.status as string)) {
    throw new Error("invalid_attribute:status");
  }
  return input as SafeTrace;
}

const result = buildSafeTrace({
  correlationId: "req-017",
  promptVersion: "support-v12",
  modelClass: "fast",
  tokensIn: 420,
  status: "validation_error",
  tenantPseudonym: "tenant-hmac-91",
});
assert.equal(result.promptVersion, "support-v12");
assert.deepEqual(Object.keys(result).sort(), [
  "correlationId", "modelClass", "promptVersion",
  "status", "tenantPseudonym", "tokensIn",
]);
assert.throws(
  () => buildSafeTrace({ ...result, prompt: "raw ticket" }),
  /sensitive_attribute:prompt/,
);
~~~
