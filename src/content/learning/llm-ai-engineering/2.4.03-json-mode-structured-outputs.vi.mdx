<CodeLessonFrame title="Parse structured output" contract="wire status/text/refusal → value | refusal | incomplete | invalid" output="'ship' enum → schema_mismatch" pitfall="Không cast JSON trực tiếp sang type hoặc biến incomplete thành value." takeaway="Validate lại required, enum, bounds và additionalProperties tại application boundary.">

`{"category":"ship"}` là JSON hợp lệ nhưng sai enum. Parser phải reject; schema-constrained output tránh nhiều lỗi nhưng không biến business authorization thành model responsibility.

~~~ts
import assert from "node:assert/strict";
type Ticket = { category: "delivery" | "refund"; urgent: boolean };
type Parsed =
  | { kind: "value"; value: Ticket }
  | { kind: "refusal"; reason: string }
  | { kind: "incomplete" }
  | { kind: "invalid"; reason: string };

function parseOutput(
  wire: { status: "completed" | "incomplete"; text?: string; refusal?: string },
): Parsed {
  if (wire.refusal) return { kind: "refusal", reason: wire.refusal };
  if (wire.status === "incomplete") return { kind: "incomplete" };
  let value: unknown;
  try { value = JSON.parse(wire.text ?? ""); }
  catch { return { kind: "invalid", reason: "invalid_json" }; }
  if (
    typeof value !== "object" || value === null ||
    Object.keys(value).sort().join(",") !== "category,urgent" ||
    !["delivery", "refund"].includes(String((value as Ticket).category)) ||
    typeof (value as Ticket).urgent !== "boolean"
  ) return { kind: "invalid", reason: "schema_mismatch" };
  return { kind: "value", value: value as Ticket };
}

assert.deepEqual(
  parseOutput({ status: "completed", text: '{"category":"delivery","urgent":false}' }),
  { kind: "value", value: { category: "delivery", urgent: false } },
);
assert.deepEqual(
  parseOutput({ status: "completed", text: '{"category":"ship","urgent":false}' }),
  { kind: "invalid", reason: "schema_mismatch" },
);
assert.equal(parseOutput({ status: "incomplete" }).kind, "incomplete");
~~~
<LessonNote>Schema adherence không thay business authorization. Production phải giữ refusal/incomplete thành nhánh riêng, validate lại tại boundary và không retry lỗi validation vô hạn.</LessonNote>

- Không xử lý output bị cắt.
- Dùng schema feature provider không hỗ trợ.
