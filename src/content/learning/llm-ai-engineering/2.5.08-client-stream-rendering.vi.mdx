~~~ts
import assert from "node:assert/strict";
type Event =
  | { type: "text_delta"; text: string }
  | { type: "completed" }
  | { type: "error"; code: string }
  | { type: "canceled" };
type ViewState = {
  status: "streaming" | "completed" | "failed" | "canceled";
  pendingText: string;
  renderedText: string;
};

function reduce(state: ViewState, event: Event): ViewState {
  if (state.status !== "streaming") throw new Error("event_after_terminal");
  if (event.type === "text_delta") {
    return { ...state, pendingText: state.pendingText + event.text };
  }
  if (event.type === "completed") {
    return {
      status: "completed", pendingText: "",
      renderedText: state.renderedText + state.pendingText,
    };
  }
  if (event.type === "canceled") return { ...state, status: "canceled" };
  return { ...state, status: "failed" };
}

function escapeText(value: string): string {
  return value.replaceAll("&", "&amp;").replaceAll("<", "&lt;").replaceAll(">", "&gt;");
}
let state: ViewState = {
  status: "streaming", pendingText: "", renderedText: "",
};
state = reduce(state, { type: "text_delta", text: "<img src=x onerror=alert(1)>" });
assert.equal(
  escapeText(state.pendingText),
  "&lt;img src=x onerror=alert(1)&gt;",
);
state = reduce(state, { type: "completed" });
assert.equal(state.status, "completed");
assert.throws(() => reduce(state, { type: "text_delta", text: "late" }), /event_after_terminal/);
~~~
