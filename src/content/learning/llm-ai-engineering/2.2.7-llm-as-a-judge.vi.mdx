<CodeLessonFrame
  title="Deterministic judge adapter"
  contract="raw JSON string → strict JudgeResult"
  output="score integer, known verdict, non-empty evidence tuple"
  pitfall="Boolean không được chấp nhận như integer score dù Python coi bool là int subclass."
  takeaway="Parser fail closed với object shape, field set, type và range không hợp lệ."
>

```python
from dataclasses import dataclass
import json

@dataclass(frozen=True)
class JudgeResult:
    score: int
    verdict: str
    evidence: tuple[str, ...]

def parse_judge(raw: str) -> JudgeResult:
    value = json.loads(raw)
    if not isinstance(value, dict):
        raise ValueError('judge output must be an object')
    if set(value) != {'score', 'verdict', 'evidence'}:
        raise ValueError('unexpected judge fields')
    score = value['score']
    if isinstance(score, bool) or not isinstance(score, int) or not 0 <= score <= 4:
        raise ValueError('score must be an integer in [0,4]')
    verdict = value['verdict']
    if not isinstance(verdict, str) or verdict not in {'pass', 'fail', 'review'}:
        raise ValueError('invalid verdict')
    evidence = value['evidence']
    if not isinstance(evidence, list) or not evidence or not all(isinstance(x, str) and x.strip() for x in evidence):
        raise ValueError('evidence must be a non-empty string list')
    return JudgeResult(score, verdict, tuple(evidence))

raw = '{"score":3,"verdict":"pass","evidence":["correctness: supported"]}'
assert parse_judge(raw) == JudgeResult(3, 'pass', ('correctness: supported',))
try:
    parse_judge('{"score":9,"verdict":"pass","evidence":["x"]}')
except ValueError as error:
    assert 'score' in str(error)
else:
    raise AssertionError('invalid score must fail closed')
for invalid in [
    '[]',
    '{"score":true,"verdict":"pass","evidence":["x"]}',
    '{"score":3,"verdict":[],"evidence":["x"]}',
]:
    try:
        parse_judge(invalid)
    except ValueError:
        pass
    else:
        raise AssertionError('invalid judge shape/type must fail closed')
```
