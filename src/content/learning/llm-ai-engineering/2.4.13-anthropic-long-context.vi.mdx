Nếu tools chiếm 20%, history 50% và documents 40% budget input thì request đã vượt 10% trước output reserve. Giải pháp là tool search/retrieval/compaction, không silent truncation.

~~~ts
import assert from "node:assert/strict";
type Budget = {
  limit: number; outputReserve: number; tools: number; history: number; documents: number;
};
function decide(parts: Budget) {
  const inputLimit = parts.limit - parts.outputReserve;
  const requestedInput = parts.tools + parts.history + parts.documents;
  const overflow = requestedInput - inputLimit;
  return {
    inputLimit,
    requestedInput,
    overflow: Math.max(0, overflow),
    decision: overflow > 0 ? "retrieve_or_compact" : "dispatch",
  } as const;
}

const decision = decide({
  limit: 10_000, outputReserve: 1_000,
  tools: 2_000, history: 5_000, documents: 4_000,
});
assert.deepEqual(decision, {
  inputLimit: 9_000, requestedInput: 11_000,
  overflow: 2_000, decision: "retrieve_or_compact",
});
~~~
