~~~ts
import assert from "node:assert/strict";

type PromptManifest = {
  id: string;
  parentId: string;
  templateHash: string;
  schemaVersion: string;
  modelConfigVersion: string;
  evalSuiteVersion: string;
  change: string;
  canaryPercent: number;
};

function publishPrompt(
  manifest: PromptManifest,
  knownIds: ReadonlySet<string>,
): Readonly<PromptManifest> {
  if (!/^support-v[1-9][0-9]*$/.test(manifest.id)) throw new Error("invalid_version_id");
  if (knownIds.has(manifest.id)) throw new Error("duplicate_version");
  if (!knownIds.has(manifest.parentId)) throw new Error("unknown_parent");
  for (const value of [
    manifest.templateHash,
    manifest.schemaVersion,
    manifest.modelConfigVersion,
    manifest.evalSuiteVersion,
    manifest.change,
  ]) {
    if (!value.trim()) throw new Error("incomplete_manifest");
  }
  if (manifest.canaryPercent < 0 || manifest.canaryPercent > 100) {
    throw new Error("invalid_canary_percent");
  }
  return Object.freeze({ ...manifest });
}

const result = publishPrompt({
  id: "support-v12",
  parentId: "support-v11",
  templateHash: "sha256:9c41",
  schemaVersion: "support-v4",
  modelConfigVersion: "fast-v3",
  evalSuiteVersion: "policy-set-7",
  change: "clarify refund wording",
  canaryPercent: 5,
}, new Set(["support-v11"]));

assert.equal(result.id, "support-v12");
assert.equal(result.parentId, "support-v11");
assert.equal(Object.isFrozen(result), true);
assert.throws(
  () => publishPrompt({
    ...result,
    id: "support-v13",
    parentId: "missing",
  }, new Set(["support-v11"])),
  /unknown_parent/,
);
assert.throws(
  () => publishPrompt({ ...result, id: "support-v11" }, new Set(["support-v11"])),
  /duplicate_version/,
);
~~~
