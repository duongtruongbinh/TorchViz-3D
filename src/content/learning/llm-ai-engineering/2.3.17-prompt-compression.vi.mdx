~~~ts
import assert from "node:assert/strict";

type PromptSection = {
  name: "policy" | "history" | "examples";
  beforeTokens: number;
  afterTokens: number;
  protected: boolean;
};

function planCompression(sections: readonly PromptSection[]) {
  if (sections.length === 0) throw new Error("empty_prompt");
  for (const section of sections) {
    if (!Number.isInteger(section.beforeTokens) || !Number.isInteger(section.afterTokens)) {
      throw new Error("token_count_must_be_integer");
    }
    if (section.beforeTokens < 0 || section.afterTokens < 0 ||
        section.afterTokens > section.beforeTokens) {
      throw new Error("invalid_token_budget");
    }
    if (section.protected && section.afterTokens !== section.beforeTokens) {
      throw new Error(`protected_section_changed:${section.name}`);
    }
  }
  const beforeTokens = sections.reduce((sum, item) => sum + item.beforeTokens, 0);
  const afterTokens = sections.reduce((sum, item) => sum + item.afterTokens, 0);
  if (beforeTokens === 0 || afterTokens >= beforeTokens) {
    throw new Error("compression_did_not_reduce_budget");
  }
  return {
    beforeTokens,
    afterTokens,
    savedTokens: beforeTokens - afterTokens,
    savingRate: (beforeTokens - afterTokens) / beforeTokens,
  };
}

const result = planCompression([
  { name: "policy", beforeTokens: 700, afterTokens: 700, protected: true },
  { name: "history", beforeTokens: 900, afterTokens: 300, protected: false },
  { name: "examples", beforeTokens: 400, afterTokens: 200, protected: false },
]);
assert.deepEqual(result, {
  beforeTokens: 2_000,
  afterTokens: 1_200,
  savedTokens: 800,
  savingRate: 0.4,
});
assert.throws(
  () => planCompression([
    { name: "policy", beforeTokens: 700, afterTokens: 600, protected: true },
  ]),
  /protected_section_changed/,
);
~~~
