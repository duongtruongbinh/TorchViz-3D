OpenAI text delta và Anthropic content-block delta cùng thành `{type:'text_delta',text}`; unknown event được telemetry nhưng không crash stream.

~~~ts
import assert from "node:assert/strict";
type InternalEvent =
  | { type: "start"; requestId: string }
  | { type: "text_delta"; requestId: string; text: string }
  | { type: "tool_delta"; requestId: string; callId: string; json: string }
  | { type: "usage"; requestId: string; inputTokens: number; outputTokens: number }
  | { type: "error"; requestId: string; code: string }
  | { type: "completed"; requestId: string };

function mapEvent(provider: "openai" | "anthropic", raw: Record<string, unknown>): InternalEvent | null {
  const requestId = String(raw.requestId);
  if (provider === "openai" && raw.type === "response.output_text.delta") {
    return { type: "text_delta", requestId, text: String(raw.delta) };
  }
  if (provider === "anthropic" && raw.type === "content_block_delta" && raw.kind === "text") {
    return { type: "text_delta", requestId, text: String(raw.text) };
  }
  if (raw.type === "completed") return { type: "completed", requestId };
  return null; // emit low-cardinality unknown-event telemetry outside payload
}

assert.deepEqual(
  mapEvent("openai", {
    type: "response.output_text.delta", requestId: "req-17", delta: "Xin",
  }),
  { type: "text_delta", requestId: "req-17", text: "Xin" },
);
assert.deepEqual(
  mapEvent("anthropic", {
    type: "content_block_delta", kind: "text", requestId: "req-17", text: " chào",
  }),
  { type: "text_delta", requestId: "req-17", text: " chào" },
);
assert.equal(mapEvent("openai", { type: "future_event", requestId: "req-17" }), null);
~~~
<LessonNote>Streaming parser phải giữ UTF-8/SSE frame chưa hoàn chỉnh, event order và AbortSignal. Unknown event telemetry không chứa raw chunk/prompt.</LessonNote>
