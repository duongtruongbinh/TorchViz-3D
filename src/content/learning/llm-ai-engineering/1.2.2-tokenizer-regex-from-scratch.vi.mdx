## Đóng gói thành hàm tokenizer

<TokenizerRegexWalkthrough content={{
  lead: { vi: 'Gom hai bước split và làm sạch vào hàm `tokenize(text)`. Từ đây, ta có thể áp dụng cùng một quy tắc cho bất kỳ chuỗi đầu vào nào mà không lặp lại code.' },
  diagram: {
    inputText: 'The robot paused, then flashed: "Package ready!"--twice.',
    outputLabel: { vi: 'Hàm tokenize(text) trả về' },
    tokens: ['The', 'robot', 'paused', ',', 'then', 'flashed', ':', '"', 'Package', 'ready', '!', '"', '--', 'twice', '.'],
  },
  code: [
    'def tokenize(text):',
    '    result = re.split(',
    '        r\'([,.:;?_!"()]|--|\\s)\',',
    '        text,',
    '    )',
    '    return [',
    '        item.strip()',
    '        for item in result',
    '        if item.strip()',
    '    ]',
    '',
    'text = \'The robot paused, then flashed: "Package ready!"--twice.\'',
    'tokens = tokenize(text)',
    'print(tokens)',
  ],
  output: [
    "['The', 'robot', 'paused', ',', 'then', 'flashed', ':', '\"', 'Package', 'ready', '!', '\"', '--', 'twice', '.']",
  ],
  takeaway: { vi: 'Hàm `tokenize` giữ một pipeline nhất quán: nhận text, split bằng regex, loại phần tử rỗng rồi trả về danh sách token.' },
}} />
