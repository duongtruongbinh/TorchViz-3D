Video support clip được upload theo policy, question là text part, output schema là damage segments. App xóa resource theo retention và không tin timestamp nếu parse fail.

~~~ts
import assert from "node:assert/strict";
type Resource = {
  id: string; tenant: string; state: "active" | "deleted"; expiresAt: number;
};
type Part =
  | { kind: "text"; text: string }
  | { kind: "file"; resourceId: string; mime: string };

function buildParts(
  instruction: string,
  resource: Resource,
  tenant: string,
  now: number,
): Part[] {
  if (resource.tenant !== tenant) throw new Error("forbidden_resource");
  if (resource.state !== "active" || resource.expiresAt <= now) {
    throw new Error("resource_unavailable");
  }
  return [
    { kind: "file", resourceId: resource.id, mime: "video/mp4" },
    { kind: "text", text: instruction },
  ];
}

const parts = buildParts(
  "Liệt kê khoảng thời gian kiện hàng bị va đập.",
  { id: "file_17", tenant: "t-a", state: "active", expiresAt: 2_000 },
  "t-a",
  1_000,
);
assert.deepEqual(parts.map(part => part.kind), ["file", "text"]);
assert.throws(
  () => buildParts(
    "inspect",
    { id: "file_17", tenant: "t-a", state: "active", expiresAt: 2_000 },
    "t-b",
    1_000,
  ),
  /forbidden_resource/,
);
~~~
<LessonNote>Upload ID là capability-scoped resource, không phải public URL. Production phải theo dõi processing/expiry/delete state và bảo toàn thứ tự parts.</LessonNote>
