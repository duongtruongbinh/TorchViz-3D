## Viết thành hàm `tokenize`

<TokenizerRegexWalkthrough content={{
  lead: { vi: 'Ta gom các bước trên vào hàm `tokenize`: tách từ và dấu câu, sau đó bỏ khoảng trắng cùng chuỗi rỗng.' },
  diagram: {
    inputText: 'Linh builds AI. Linh tests AI.',
    outputLabel: { vi: 'tokenize(text) trả về' },
    tokens: ['Linh', 'builds', 'AI', '.', 'Linh', 'tests', 'AI', '.'],
  },
  code: [
    'def tokenize(text: str) -> list[str]:',
    '    parts = re.split(r"([,.:;?_!\\\"()]|--|\\s)", text)',
    '    return [part.strip() for part in parts if part.strip()]',
    '',
    'tokens = tokenize(raw_text)',
    'print(tokens)',
  ],
  output: ["['Linh', 'builds', 'AI', '.', 'Linh', 'tests', 'AI', '.']"],
  takeaway: { vi: 'Kết quả là một danh sách token theo đúng thứ tự xuất hiện trong câu. Trong ví dụ này, dấu câu được giữ lại còn khoảng trắng được bỏ đi.' },

  code: [
    'import re',

    'def tokenize(text: str) -> list[str]:',

    '    return [part.strip() for part in parts if part.strip()]',
    '',
    'paragraph = (',
    '    "At dawn, Linh opened the workshop. "',
    "    'The robot flashed: \\\"Package ready!\\\"--twice.'",
    ')',
    '',
    'paragraph_tokens = tokenize(paragraph)',
    'print(paragraph_tokens)',
    '',
    'assert paragraph_tokens[:6] == [',
    '    "At", "dawn", ",", "Linh", "opened", "the"',
    ']',
    "assert '\\\"' in paragraph_tokens",
    'assert "--" in paragraph_tokens',
    'assert all(token.strip() == token and token for token in paragraph_tokens)',
  ],
