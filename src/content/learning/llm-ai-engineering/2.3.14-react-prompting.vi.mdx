~~~ts
import assert from "node:assert/strict";

type ToolProposal = { callId: string; tool: string; args: unknown };
type ToolResult = {
  kind: "tool_result";
  callId: string;
  value: { orderId: string; status: string };
};
type AuthContext = {
  allowedTools: ReadonlySet<string>;
  ownedOrders: ReadonlySet<string>;
  maxSteps: number;
};

const orderStatus: Readonly<Record<string, string>> = { A17: "late" };

function executeProposal(
  proposal: ToolProposal,
  context: AuthContext,
  step: number,
): ToolResult {
  if (!Number.isInteger(step) || step < 1 || step > context.maxSteps) {
    throw new Error("tool_loop_limit");
  }
  if (!/^call-[a-z0-9]+$/.test(proposal.callId)) throw new Error("invalid_call_id");
  if (!context.allowedTools.has(proposal.tool)) throw new Error("tool_not_allowed");
  if (proposal.tool !== "lookup_order") throw new Error("unsupported_tool");
  if (!proposal.args || typeof proposal.args !== "object" || Array.isArray(proposal.args)) {
    throw new Error("invalid_tool_args");
  }
  const args = proposal.args as Record<string, unknown>;
  if (Object.keys(args).length !== 1 || typeof args.orderId !== "string") {
    throw new Error("invalid_tool_args");
  }
  if (!context.ownedOrders.has(args.orderId)) throw new Error("order_not_owned");
  const status = orderStatus[args.orderId];
  if (!status) throw new Error("order_not_found");
  return {
    kind: "tool_result",
    callId: proposal.callId,
    value: { orderId: args.orderId, status },
  };
}

const context: AuthContext = {
  allowedTools: new Set(["lookup_order"]),
  ownedOrders: new Set(["A17"]),
  maxSteps: 3,
};
assert.deepEqual(
  executeProposal(
    { callId: "call-a1", tool: "lookup_order", args: { orderId: "A17" } },
    context,
    1,
  ),
  {
    kind: "tool_result",
    callId: "call-a1",
    value: { orderId: "A17", status: "late" },
  },
);
assert.throws(
  () => executeProposal({ callId: "call-a2", tool: "refund_order", args: { orderId: "A17" } }, context, 2),
  /tool_not_allowed/,
);
assert.throws(
  () => executeProposal({ callId: "call-a3", tool: "lookup_order", args: { orderId: "A18" } }, context, 2),
  /order_not_owned/,
);
assert.throws(
  () => executeProposal({ callId: "call-a4", tool: "lookup_order", args: { orderId: "A17" } }, context, 4),
  /tool_loop_limit/,
);
~~~
