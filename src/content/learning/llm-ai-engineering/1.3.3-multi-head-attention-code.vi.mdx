```python
import torch
from torch import nn

class CausalMultiHeadAttention(nn.Module):
    def __init__(self, d_model, n_heads, dropout=0.0):
        super().__init__()
        if d_model <= 0 or n_heads <= 0:
            raise ValueError("d_model and n_heads must be positive")
        if d_model % n_heads:
            raise ValueError("d_model must be divisible by n_heads")
        self.n_heads = n_heads
        self.head_dim = d_model // n_heads
        self.qkv = nn.Linear(d_model, 3 * d_model, bias=False)
        self.out_proj = nn.Linear(d_model, d_model, bias=False)
        self.dropout = nn.Dropout(dropout)
```

Một linear chung tạo `(B,T,3C)`. Production thường cache mask và dùng kernel fused.

```python
import torch
from torch import nn

class CausalMultiHeadAttention(nn.Module):
    def __init__(self, d_model, n_heads, dropout=0.0):
        super().__init__()
        if d_model <= 0 or n_heads <= 0:
            raise ValueError("d_model and n_heads must be positive")
        if d_model % n_heads:
            raise ValueError("d_model must be divisible by n_heads")
        self.n_heads = n_heads
        self.head_dim = d_model // n_heads
        self.qkv = nn.Linear(d_model, 3 * d_model, bias=False)
        self.out_proj = nn.Linear(d_model, d_model, bias=False)
        self.dropout = nn.Dropout(dropout)

    def forward(self, x):
        if x.ndim != 3:
            raise ValueError("x must have shape (B,T,C)")
        B, T, C = x.shape
        if T == 0 or C != self.n_heads * self.head_dim:
            raise ValueError("x has an invalid sequence or feature shape")
        q, k, v = self.qkv(x).chunk(3, dim=-1)
        def split(t):
            return t.view(B, T, self.n_heads, self.head_dim).transpose(1, 2)
        q, k, v = map(split, (q, k, v))
        scores = q @ k.transpose(-2, -1) / (self.head_dim ** 0.5)
        mask = torch.triu(
            torch.ones(T, T, device=x.device, dtype=torch.bool), 1
        )
        weights = torch.softmax(
            scores.masked_fill(mask, -torch.inf), dim=-1
        )
        context = self.dropout(weights) @ v
        context = context.transpose(1, 2).contiguous().view(B, T, C)
        return self.out_proj(context)

attention = CausalMultiHeadAttention(128, 8, dropout=0.0)
x = torch.randn(2, 16, 128)
y = attention(x)
assert y.shape == x.shape
assert sum(p.numel() for p in attention.parameters()) == 4 * 128 * 128

attention.eval()
x_changed = x.clone()
x_changed[:, -1, :] += 100.0
assert torch.allclose(
    attention(x)[:, :-1],
    attention(x_changed)[:, :-1],
    atol=1e-5,
)

try:
    CausalMultiHeadAttention(10, 3)
except ValueError:
    pass
else:
    raise AssertionError("invalid head split must fail")

print(y.shape)  # torch.Size([2, 16, 128])
```
