## Thiết lập Không gian Siêu tham số & ASHA với Ray Tune

Tạo file `madewithml/tune.py` định nghĩa không gian tìm kiếm:

```python
from ray import tune
from ray.tune.schedulers import ASHAScheduler
from ray.train.torch import TorchTrainer
from ray.train import ScalingConfig

def tune_hyperparameters(train_ds, num_samples: int = 8):
    """Tìm kiếm siêu tham số tự động với ASHA Scheduler."""
    
    # 1. Định nghĩa không gian tìm kiếm
    param_space = {
        "train_loop_config": {
            "num_classes": 4,
            "dropout_p": tune.uniform(0.2, 0.6),
            "lr": tune.loguniform(1e-5, 5e-4),
            "batch_size": tune.choice([64, 128, 256]),
            "num_epochs": 10,
        }
    }
    
    # 2. Khởi tạo ASHA Early Stopping Scheduler
    scheduler = ASHAScheduler(
        max_t=10,
        grace_period=2, # Chạy tối thiểu 2 epochs trước khi xét prune
        reduction_factor=3,
    )
    
    # 3. Chạy quá trình Tuning
    tuner = tune.Tuner(
        TorchTrainer(
            train_loop_per_worker=train_loop_per_worker,
            scaling_config=ScalingConfig(num_workers=1, use_gpu=False),
            datasets={"train": train_ds},
        ),
        param_space=param_space,
        tune_config=tune.TuneConfig(
            metric="val_loss",
            mode="min",
            scheduler=scheduler,
            num_samples=num_samples,
        ),
    )
    
    results = tuner.fit()
    best_result = results.get_best_result(metric="val_loss", mode="min")
    print(f"✨ Cấu hình tốt nhất: {best_result.config['train_loop_config']}")
    return best_result
```

## Xây dựng Hàm Đánh giá Lát cắt (Sliced Evaluation)

Tạo file `madewithml/evaluate.py` để tính toán metrics theo từng nhóm nghiệp vụ:

```python
import numpy as np
import pandas as pd
from sklearn.metrics import precision_recall_fscore_support

def evaluate_slices(df: pd.DataFrame, y_true: np.ndarray, y_pred: np.ndarray) -> dict:
    """Đo lường hiệu năng trên các lát cắt dữ liệu đặc thù."""
    slice_metrics = {}
    
    # 1. Lát cắt theo độ dài tiêu đề (Ngắn vs Dài)
    short_mask = df["text"].apply(lambda t: len(t.split()) < 8).values
    p, r, f1, _ = precision_recall_fscore_support(
        y_true[short_mask], y_pred[short_mask], average="weighted", zero_division=0
    )
    slice_metrics["short_text"] = {"f1": float(f1), "precision": float(p), "recall": float(r), "count": int(short_mask.sum())}
    
    long_mask = ~short_mask
    p, r, f1, _ = precision_recall_fscore_support(
        y_true[long_mask], y_pred[long_mask], average="weighted", zero_division=0
    )
    slice_metrics["long_text"] = {"f1": float(f1), "precision": float(p), "recall": float(r), "count": int(long_mask.sum())}
    
    return slice_metrics
```

## Trực quan hóa Ma trận Nhầm lẫn & Báo cáo

Kết hợp các chỉ số tổng thể và xuất báo cáo hoàn chỉnh:

```python
from sklearn.metrics import classification_report, confusion_matrix
import json

def generate_evaluation_report(y_true: list, y_pred: list, classes: list[str], output_path: str):
    """Xuất báo cáo đánh giá và ma trận nhầm lẫn chuẩn hóa."""
    
    # 1. Classification report
    report = classification_report(y_true, y_pred, target_names=classes, output_dict=True)
    
    # 2. Normalized Confusion Matrix
    cm = confusion_matrix(y_true, y_pred, normalize="true")
    
    full_results = {
        "overall": report,
        "confusion_matrix": cm.tolist(),
    }
    
    with open(output_path, "w") as f:
        json.dump(full_results, f, indent=2)
        
    print(f"📊 Báo cáo đánh giá đã được lưu tại: {output_path}")
    print(f"Overall Weighted F1: {report['weighted avg']['f1-score']:.4f}")
```
