## Định nghĩa Kiến trúc Mạng Nơ-ron PyTorch

Tạo module `madewithml/models.py` định nghĩa lớp nơ-ron kết hợp pre-trained BERT và phân loại:

```python
import torch
import torch.nn as nn
from transformers import BertModel

class FinBertClassifier(nn.Module):
    """Mô hình phân loại văn bản sử dụng BERT backbone."""
    
    def __init__(self, bert_model_name: str = "google/bert_uncased_L-2_H-128_A-2", num_classes: int = 4, dropout_p: float = 0.5):
        super().__init__()
        self.bert = BertModel.from_pretrained(bert_model_name)
        self.dropout = nn.Dropout(dropout_p)
        self.fc = nn.Linear(self.bert.config.hidden_size, num_classes)

    def forward(self, input_ids: torch.Tensor, attention_mask: torch.Tensor) -> torch.Tensor:
        outputs = self.bert(input_ids=input_ids, attention_mask=attention_mask)
        # Lấy vector đại diện [CLS] token
        cls_output = outputs.last_hidden_state[:, 0, :]
        dropout_output = self.dropout(cls_output)
        logits = self.fc(dropout_output)
        return logits
```

## Xây dựng Hàm Huấn luyện Phân tán Train Loop

Hàm `train_loop_per_worker` sẽ được Ray Train thực thi đồng thời trên từng worker:

```python
import ray.train
import ray.train.torch
import torch.optim as optim

def train_loop_per_worker(config: dict):
    """Vòng lặp huấn luyện thực thi trên từng worker GPU/CPU."""
    # 1. Khởi tạo mô hình và bọc bằng Ray Torch DDP
    model = FinBertClassifier(
        num_classes=config["num_classes"],
        dropout_p=config["dropout_p"]
    )
    model = ray.train.torch.prepare_model(model)
    
    criterion = nn.CrossEntropyLoss()
    optimizer = optim.AdamW(model.parameters(), lr=config["lr"])
    
    # 2. Lấy shard dataset phân tán cho worker hiện tại
    train_ds = ray.train.get_dataset_shard("train")
    
    # 3. Vòng lặp các Epoch
    for epoch in range(config["num_epochs"]):
        model.train()
        total_loss = 0.0
        
        for batch in train_ds.iter_torch_batches(batch_size=config["batch_size"]):
            optimizer.zero_grad()
            logits = model(batch["ids"], batch["masks"])
            loss = criterion(logits, batch["targets"])
            loss.backward()
            optimizer.step()
            total_loss += loss.item()
            
        # 4. Báo cáo metrics về Ray Train Driver
        ray.train.report(metrics={"epoch": epoch, "train_loss": total_loss})
```

## Khởi chạy ScalingConfig với Ray Train

Cấu hình tài nguyên tính toán và kích hoạt phân tán qua `TorchTrainer`:

```python
from ray.train.torch import TorchTrainer
from ray.train import ScalingConfig

def run_distributed_training(train_ds, num_workers: int = 2, use_gpu: bool = False):
    """Thiết lập TorchTrainer và thực thi huấn luyện song song."""
    trainer = TorchTrainer(
        train_loop_per_worker=train_loop_per_worker,
        train_loop_config={
            "num_classes": 4,
            "dropout_p": 0.5,
            "lr": 1e-4,
            "batch_size": 128,
            "num_epochs": 5,
        },
        scaling_config=ScalingConfig(
            num_workers=num_workers,
            use_gpu=use_gpu,
        ),
        datasets={"train": train_ds},
    )
    
    results = trainer.fit()
    print("🎉 Huấn luyện phân tán hoàn tất!")
    print(f"Metrics tốt nhất: {results.metrics}")
    return results
```
